```python
"""Task driver (11-spec §5): the kernel-side round loop for armed tasks.

The driver owns the generic lifecycle of an armed task: ask the profile for
the next round, run it through the ordinary turn path, let the profile
settle it mechanically, stop on completion / a failing round / the round
cap / a disarm — and leave accounting to the profile's hooks.  Everything
plan-specific lives behind ``RoundProfile``, so one driver serves `/plan`
today and any future armed task, with the plan behaviour pinned by the
existing plan suites (its Prove).
"""

from __future__ import annotations

from typing import Any, Protocol

__all__ = ["RoundProfile", "TaskDriver"]


class RoundProfile(Protocol):
    """What one kind of armed task must provide; one hook per existing part.

    Every method maps 1:1 to a piece the plan runner already had (11-spec
    §6.1), so the plan profile is an adapter, not a rewrite.
    """

    def new_report(self) -> Any: ...

    def max_rounds(self) -> int: ...

    def prepare(self, session: Any, task: Any) -> set[str]: ...

    def before_round(self, task: Any, session: Any, report: Any, receipted: set[str]) -> None: ...

    def next_item(self, task: Any, attempted: set[str]) -> Any | None: ...

    def no_item(self, task: Any, report: Any) -> None: ...

    def rounds_used(self, report: Any) -> int: ...

    def budget_stop(self, report: Any, cap: int) -> None: ...

    async def run_round(
        self, session: Any, task: Any, item: Any, report: Any, console: Any,
        *, on_delta: Any, observer: Any,
    ) -> None: ...

    def after_round(self, session: Any, task: Any, report: Any) -> None: ...

    def should_stop(self, report: Any, item: Any) -> bool: ...

    def taskless(self, report: Any, console: Any) -> None: ...

    def persist(self, session: Any) -> None: ...

    def finish(self, console: Any, report: Any) -> None: ...


class TaskDriver:
    """Drive one armed task round by round (11-spec §5.1)."""

    def __init__(self, loop: Any) -> None:
        self._loop = loop

    async def run(
        self,
        session: Any,
        profile: RoundProfile,
        console: Any = None,
        *,
        on_delta: Any = None,
        observer: Any = None,
    ) -> Any:
        report = profile.new_report()
        task = getattr(session, "task", None)
        if task is None:
            # No task at all: same verdict as an empty plan — "nothing to
            # do" — and no persist, matching the plan runner's early exit.
            profile.taskless(report, console)
            return report
        attempted: set[str] = set()
        cap = profile.max_rounds()
        receipted = profile.prepare(session, task)
        while True:
            profile.before_round(task, session, report, receipted)
            item = profile.next_item(task, attempted)
            if item is None:
                profile.no_item(task, report)
                break
            if 0 < cap <= profile.rounds_used(report):
                profile.budget_stop(report, cap)
                break
            attempted.add(item.id)
            await profile.run_round(
                session, task, item, report, console,
                on_delta=on_delta, observer=observer,
            )
            receipted.add(item.id)
            profile.after_round(session, task, report)
            if profile.should_stop(report, item):
                break
        profile.persist(session)
        profile.finish(console, report)
        return report
```

```python
def find_armed_session(sessions: Any) -> Any | None:
    """The newest persisted session whose task is armed (11-spec §7.2).

    Scanned from the raw JSON keys, not ``list_sessions`` -- this runs before
    any session UI, and the message bodies are not needed to pick a winner.
    ``openTurn`` is deliberately NOT part of the filter: a clean close clears
    it, and a crash between rounds leaves exactly that state.
    """
    root = Path(getattr(sessions, "root", "") or "")
    if not root.is_dir():
        return None
    best: tuple[str, str] | None = None
    for path in root.glob("*.json"):
        try:
            raw = json.loads(path.read_text(encoding="utf-8"))
        except (OSError, ValueError):
            continue
        task = raw.get("task")
        if not isinstance(task, dict) or task.get("armed") is not True:
            continue
        stamp = str(raw.get("lastAccessTime", "") or "")
        if best is None or stamp > best[0]:
            best = (stamp, path.stem)
    if best is None:
        return None
    return sessions.load(best[1])
```

    def _launch_boot_resume(world: Any, service: Any, bridge: Any, session: Any) -> None:
        """Resume an armed task after a restart (11-spec §7.2): the interrupted
        round first (when one is open), then the driver's own loop."""

        async def run_resume() -> None:
            from ai_coding.core.plan_runner import PlanRunner
            from ai_coding.core.task_driver import TaskDriver

            task = getattr(session, "task", None)
            # Count BEFORE resuming: if this resume crashes too, the count is
            # already on disk and the breaker keeps converging.
            if task is not None:
                task.resume_count = int(getattr(task, "resume_count", 0) or 0) + 1
                _persist_session(world, session)
            if session.open_turn is not None:
                await world.loop.resume_turn(session, on_delta=bridge.on_delta, observer=bridge)
            if not getattr(task, "armed", False):
                return
            console = IpcConsole(
                lambda method, message: service.notify(method, message), bridge.on_delta
            )
            await TaskDriver(world.loop).run(
                session, PlanRunner(world.loop), console,
                on_delta=bridge.on_delta, observer=bridge,
            )

        _launch_turn(run_resume)
